[template public genAttributes(aClassifier : uml::Classifier, modular : Boolean)]
  [for (anAttribute : uml::Property | aClassifier.u_attributes())]
    "[anAttribute.name/]": {
      "@id": "[anAttribute.iri()/]",
    [if (anAttribute.type.oclIsTypeOf(uml::Enumeration))]
        "@type": "@vocab",
        "@context": {
        [for (aLiteral : uml::EnumerationLiteral | anAttribute.type.oclAsType(uml::Enumeration).ownedLiteral) separator(',\n  ')]
            "[aLiteral.name/]": "[aLiteral.iri()/]"[/for]
        }
    [elseif (anAttribute.type.isComplex())]
        "@type": "@id"[if (modular)],[/if]
      [if (modular)]
          "@context": "[anAttribute.type.name/].jsonld"
      [/if]
    [else]
        "@type": "[anAttribute.type.iri()/]"
    [/if]
    },
  [/for]
[/template]

[template public genAttributesProfile(
  aClassifier : uml::Classifier,
  modular : Boolean,
  aClass : uml::Class,
  aProfilePackage : uml::Package)]
  [for (anAttribute : uml::Property | aClassifier.u_attributes()
      ->select(a | a.u_isAttributeInProfile(aClass, aProfilePackage)))]
    "[anAttribute.name/]": {
      "@id": "[anAttribute.iri()/]",
    [if (anAttribute.type.oclIsTypeOf(uml::Enumeration))]
        "@type": "@vocab",
        "@context": {
        [for (aLiteral : uml::EnumerationLiteral | anAttribute.type.oclAsType(uml::Enumeration).ownedLiteral) separator(',\n  ')]
            "[aLiteral.name/]": "[aLiteral.iri()/]"[/for]
        }
    [elseif (anAttribute.type.isComplex())]
        "@type": "@id"[if (modular)],[/if]
      [if (modular)]
          "@context": "[anAttribute.type.name/].jsonld"
      [/if]
    [else]
        "@type": "[anAttribute.type.iri()/]"
    [/if]
    },
  [/for]
[/template]
