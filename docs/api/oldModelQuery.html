module ucmism2t::templates::query::oldModelQuery

Module contains all model queries. Queries with prefix "e_" are migrated from Acceleo 3.7. Queries with prefix "u_" are written for Acceleo 4.
Metamodels http://www.eclipse.org/uml2/5.0.0/UML, http://www.eclipse.org/uml2/5.0.0/Types, http://www.eclipse.org/uml2/5.0.0/UML/Profile/Standard

[query public e_modelAcronym(aModel : uml::Model) : String =
	let fromProperties : String =
		'MODEL_ACRONYM'.getPropertyValue()
	in
	let fromModel : String =
		aModel.eAllContents(uml::DataType)
			->select(temp63 | temp63.name = 'ModelIdentification')
			->first()
			.eContents(uml::Property)
			->select(temp64 | temp64.name = 'acronym')
			->first().default
	in
	if fromProperties.size() > 0
	then fromProperties
	else
		if not (fromModel = null)
		then fromModel
		else aModel.name
		endif
	endif
/]
