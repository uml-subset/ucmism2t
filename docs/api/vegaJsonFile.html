[template public classJson(aClass : uml::Class)]
  [let targetClassList : OrderedSet(uml::Class) =
  	aClass.u_allSuperClasses()

  	->collect(c1 | c1.u_outgoingAssociatedClasses())
  	->sortedBy(c2 | c2.name)
  	]
    {
      "name": "[aClass.nodeName()/]"[if (targetClassList->size() > 0)]
