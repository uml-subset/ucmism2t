module ucmism2t::queries::classifierQuery

Module contains queries with a UML classifier as receiver, applicable to both calsses and data types.

[query public Xu_subClassesOrSubDataTypes(
		aClassifier : uml::Classifier) : OrderedSet(uml::Classifier) =
	if aClassifier.oclIsTypeOf(uml::Class)
	then
		aClassifier.u_subClasses()
	else
		if aClassifier.oclIsKindOf(uml::DataType)
		then
		 aClassifier.u_subDataTypes()
		else OrderedSet{}
		endif
	endif

[query public u_subClassesOrSubDataTypes(
  aClassifier : uml::Classifier) : OrderedSet(uml::Classifier) =
  if aClassifier.oclIsKindOf(uml::Class)
  then
    aClassifier.oclAsType(uml::Class).u_subClasses()
  else
    if aClassifier.oclIsKindOf(uml::DataType)
    then
      aClassifier.oclAsType(uml::DataType).u_subDataTypes()
    else
      OrderedSet{}
    endif
  endif
/]

AQL collection operations sortedBy() collect() filter() select()
Show source 
[query public u_attributes(
  aClassifier : uml::Classifier) : OrderedSet(uml::Property) =
  aClassifier->filter({uml::Class | uml::DataType})
    ->collect(c | c.ownedAttribute
      ->select(p | p.association = null))
    ->sortedBy(p | p.name)
/]
