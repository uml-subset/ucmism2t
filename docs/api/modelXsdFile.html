[template public TprocessModelXsd(aModel : uml::Model)]
  [file (aModel.e_modelShortName() + '/encoding/xml-schema/' + aModel.e_modelAcronym().toLower() + '.xsd', overwrite, 'UTF-8')]
    [for (aClass : uml::Class | aModel.u_mainPackage().eAllContents(uml::Class)->sortedBy(temp1 | temp1.name))]
      [aClass.classXsd()/]
    [/for]
  [/file]

[template public processModelXsd(aModel : uml::Model)]
  [file (aModel.e_modelShortName() + '/encoding/xml-schema/' + aModel.e_modelAcronym().toLower() + '.xsd', overwrite, 'UTF-8')]
    [aModel.xsdHeader()/]
    [if ('DDI-CDI'.isPropertyTrue())][aModel.xsdWrapper()/][/if]
    [for (aClass : uml::Class | aModel.u_mainPackage().eAllContents(uml::Class)->sortedBy(temp2 | temp2.name))]
      [aClass.classXsd()/]
    [/for]
    [for (anEnumeration : uml::Enumeration | aModel.u_mainPackage().eAllContents(uml::Enumeration)->sortedBy(temp3 | temp3.name))]
      [anEnumeration.enumerationXsd()/]
    [/for]
    [for (aDataType : uml::DataType | aModel.u_mainPackage().eAllContents(uml::DataType)->sortedBy(temp4 | temp4.name)->select(temp5 | not temp5.oclIsTypeOf(uml::Enumeration))->select(temp6 | not temp6.qualifiedName.contains('XMLSchemaDataTypes')))]
      [aDataType.dataTypeXsd()/]
    [/for]
    [aModel.xsdFooter()/]
  [/file]
[/template]
