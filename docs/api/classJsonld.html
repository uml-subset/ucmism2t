[template public classJsonldCore(aClass : uml::Class, modular : Boolean)]
  [aClass.genAttributes(modular)/]
  [for (anAssociation : uml::Association | aClass.u_associations()
    ->select(anAssociation | anAssociation.u_ownedEndClass() = aClass)
    ->sortedBy(a | a.shortName())
    )]
    "[anAssociation.shortName()/]": {
      "@id": "[anAssociation.iri()/]",
      "@type": "@id"[if (modular)],[/if]
    [if (modular)]
        "@context": "[anAssociation.u_otherEndClass().name/].jsonld"
    [/if]
    },
  [/for]
  [for (anAssociation : uml::Association | aClass.u_associations()
    ->select(a1 | a1.u_otherEndClass() = aClass)
    ->sortedBy(a2 | a2.shortName())
    ->sortedBy(a3 | a3.u_ownedEndClass().name )
    )]
    "[anAssociation.shortName()/]_OF_[anAssociation.u_ownedEndClass().name/]": {
      "@reverse": "[anAssociation.iri()/]",
      "@type": "@id"[if (modular)],[/if]
    [if (modular)]
        "@context": "[anAssociation.u_ownedEndClass().name/].jsonld"
    [/if]
    },
  [/for]

[template public classJsonldCoreProfile(
  aClass : uml::Class,
  modular : Boolean,
  aProfilePackage : uml::Package)]
  [let subsetClasses : OrderedSet(uml::Class) =
      aProfilePackage.u_childDependencySupplierClasses()]
    [aClass.genAttributesProfile(modular, aClass, aProfilePackage)/]
    [for (anAssociation : uml::Association | aClass.u_associations()
        ->select(a | a.u_ownedEndClass() = aClass
          and subsetClasses->includes(a.u_ownedEndClass())
          and subsetClasses->includes(a.u_otherEndClass()))
        ->sortedBy(a | a.shortName()))]
      "[anAssociation.shortName()/]": {
        "@id": "[anAssociation.iri()/]",
        "@type": "@id"[if (modular)],[/if]
      [if (modular)]
          "@context": "[anAssociation.u_otherEndClass().name/].jsonld"
      [/if]
      },
    [/for]
    [for (anAssociation : uml::Association | aClass.u_associations()
        ->select(a | a.u_otherEndClass() = aClass
          and subsetClasses->includes(a.u_ownedEndClass())
          and subsetClasses->includes(a.u_otherEndClass()))
        ->sortedBy(a | a.shortName())
        ->sortedBy(a | a.u_ownedEndClass().name))]
      "[anAssociation.shortName()/]_OF_[anAssociation.u_ownedEndClass().name/]": {
        "@reverse": "[anAssociation.iri()/]",
        "@type": "@id"[if (modular)],[/if]
      [if (modular)]
          "@context": "[anAssociation.u_ownedEndClass().name/].jsonld"
      [/if]
      },
    [/for]
    " comment ": "tag:json-should-support-trailing-commas"
  [/let]
[/template]

[template public classJsonldSelfContainedProfile(
  aClass : uml::Class,
  aProfilePackage : uml::Package)]
  [let subsetClasses : OrderedSet(uml::Class) =
      aProfilePackage.u_childDependencySupplierClasses()]
    [if (aClass.superClass->notEmpty()
        and aClass.superClass->asSequence()->first().isNotInMappedItems()
        and subsetClasses->includes(aClass.superClass->asSequence()->first()))]
      [aClass.superClass->asSequence()->first().classJsonldSelfContainedProfile(aProfilePackage)/],
    [/if]
    {
      [aClass.classJsonldCoreProfile(false, aProfilePackage)/]
    }
  [/let]
[/template]
