module ucmism2t::queries::eobjectQuery

Module contains queries with ecore::EObject as receiver, used as an entry point for accessing global configuration values that are not tied to any specific UML element. Includes queries for RDF URI mapping tokens (uml-map-rdf property) and test package names.
Metamodels http://www.eclipse.org/uml2/5.0.0/UML, http://www.eclipse.org/uml2/5.0.0/Types, http://www.eclipse.org/uml2/5.0.0/UML/Profile/Standard
Sourceucmism2t/queries/eobjectQuery.mtl

[query public u_mappedUcmisItemsRdfTokens(
		anEObject : ecore::EObject) : Sequence(String) =
	'uml-map-rdf'.getPropertyValue()
		.replaceAll('\\s+', '')
		.tokenize(',')
/]

[query public u_mappedUcmisItemsRdf(
		anEObject : ecore::EObject) : Sequence(String) =
	anEObject.u_mappedUcmisItemsRdfTokens()
	  ->select(s1 | s1.contains('>'))
		->collect(s2 | s2.tokenize('>')->at(1))

[query public u_testPackages(anEObject : ecore::EObject) : OrderedSet(String) =
  'TEST_PACKAGE'.getPropertyValue()
    .substituteAll(' ', '')
    .tokenize(',')
    ->asOrderedSet()
