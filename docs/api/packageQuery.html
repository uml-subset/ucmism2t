[query public u_isAllowedPackage(aPackage : uml::Package) : Boolean =
  let mainPkg : uml::Package = aPackage.u_model().u_mainPackage() in
  aPackage = mainPkg
  or (mainPkg.eAllContents(uml::Package)->includes(aPackage)
      and aPackage.u_isNotIgnoredPackage())
/]

[query public u_isNotIgnoredPackage(aPackage : uml::Package) : Boolean =
  'IGNORE_PACKAGE'.getPropertyValue()
    .substituteAll(' ', '')
    .tokenize(',')
    ->collect(p | aPackage.u_model().u_namedElementByName(p))
    ->excludes(aPackage)
