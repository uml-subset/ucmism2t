[query public u_childDependencySupplierClasses(

[query public u_profileSubsetClasses(
  anElement : uml::NamedElement) : OrderedSet(uml::Class) =
  if not 'profiles'.isPropertyTrue()
  then OrderedSet{}
  else
    let profilesPackage : uml::Package =
      anElement.u_model().eContents(uml::Package)
        ->select(p | p.name = 'profiles_package'.getRequiredPropertyValue())
        ->first()
    in
    if profilesPackage = null
    then OrderedSet{}
    else
      profilesPackage.eContents(uml::Package)
        ->collect(subPkg | subPkg.u_childDependencySupplierClasses())
        ->asOrderedSet()
    endif
  endif

query public u_selectedAttributes

[query public u_selectedAttributes(aClass : uml::Class, aProfilePackage : uml::Package) : Sequence(String) = ...]
Returns the sequence of selected attribute names for the given class within the given profile package, as defined by the @selected-attributes annotation on the dependency relationship from the profile package to the class. The annotation appears in the body of an ownedComment on the dependency element, on a line of the form: @selected-attributes: attr1, attr2, attr3 Multiple annotations may appear in the comment body on separate lines; only the @selected-attributes line is processed here. When the annotation is present, only the listed attribute names should be included in profile-specific documentation and encodings. When the annotation is absent (the dependency exists but has no such annotation), an empty sequence is returned — callers must interpret an empty result as meaning all attributes are included (no restriction). Attribute names are literal UML property name values as they appear in the model, including any CURIE prefix (e.g. 'schema:identifier', 'prov:wasDerivedFrom') used by convention for attributes from external namespaces.

aClass : uml::Class
- The UML class whose selected attributes are to be retrieved.
aProfilePackage : uml::Package
- The profile sub-package that defines the subset.

Returns

Sequence(String) — A Sequence of selected attribute name strings, or an empty Sequence if no @selected-attributes annotation is present.

AQL services eContents() and() contains() tokenize() or() substituteAll() substring() index()
Show source 
[query public u_selectedAttributes(
  aClass : uml::Class,
  aProfilePackage : uml::Package) : Sequence(String) =
  let dep : uml::Dependency =
    aProfilePackage.eContents(uml::Dependency)
      ->select(d |
        d.client->filter(uml::Package)->first() = aProfilePackage
        and d.supplier->includes(aClass))
      ->first()
  in
  if dep = null
  then Sequence{}
  else
    let annotatedComment : uml::Comment =
      dep.ownedComment
        ->select(c | c.body <> null
          and c.body.contains('@selected-attributes'))
        ->first()
    in
    if annotatedComment = null
    then Sequence{}
    else
      let line : String =
        annotatedComment.body
          .tokenize('\n')
          ->select(l | l.contains('@selected-attributes'))
          ->first()
      in
      if line = null or line = ''
      then Sequence{}
      else
        line.substring(line.index(':') + 1)
          .substituteAll(' ', '')
          .tokenize(',')
      endif
    endif
  endif
/]
