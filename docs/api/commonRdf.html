[template public iri(x : uml::NamedElement) post(self.cdiAdHoc())]
  [if (x.ancestors()->notEmpty())]
    [if (x.oclIsTypeOf(uml::Property))]
      		[x.u_parent().oclAsType(uml::Classifier).name/]-[x.name/]
    [else]
      	    [x.name/]
    [/if]
  [else]
    	[x.builtinIri()/]
  [/if]
[/template]

[template public cdiAdHoc(txt : String) post(self.noWhitespace())]
  [let iri : String = txt.iriEscape()]
    [if (txt.u_mappedUcmisItemsRdf()->indexOf(iri) = 0)]
      		['RDF_PREFIX'.getPropertyValue()/]:[iri/]
    [else]
      		[txt.u_mappedUcmisItemsRdfTargets()->at(txt.u_mappedUcmisItemsRdf()->indexOf(iri))/]
    [/if]
  [/let]
[/template]
