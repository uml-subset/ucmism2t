[template public classXsd(aClass : uml::Class)]
    <xs:element name="[aClass.name/]"
                type="[aClass.name/]XsdType"
                xml:id="[aClass.name/]">
      <!-- based on the UML class [aClass.qualifiedName/] -->
      <xs:annotation>
  ['      <xs:documentation>' + aClass.u_commentBody().xsd_escapeXmlCharacters()/]</xs:documentation>
      </xs:annotation>
    </xs:element>
    <xs:complexType name="[aClass.name/]XsdType"
                    xml:id="[aClass.name/]XsdType">
      <xs:annotation>
  ['      <xs:documentation>' + aClass.u_commentBody().xsd_escapeXmlCharacters()/]</xs:documentation>
      </xs:annotation>
      [if (aClass.u_allSuperClasses()->notEmpty())]
        <xs:complexContent>
          <xs:extension base="[aClass.u_allSuperClasses()->first().name/]XsdType">
            [if (aClass.u_attributes()->notEmpty() or aClass.u_associationEndProperties()->notEmpty())]
              [aClass.propertyXsd()/]
            [/if]
          </xs:extension>
        </xs:complexContent>
      [else]
        [if (aClass.u_attributes()->notEmpty() or aClass.u_associationEndProperties()->notEmpty())]
          [aClass.propertyXsd()/]
        [/if]
      [/if]
    </xs:complexType>
[/template]

[template public propertyXsd(aClass : uml::Class)]
  <xs:sequence>
    [for (anAttribute : uml::Property | aClass.u_attributes())]
      <xs:element name="[anAttribute.name/]"
                  type="[anAttribute.type.xsd_typeName()/]"
                  minOccurs="[anAttribute.lower/]"
                  maxOccurs="[anAttribute.xsd_upperCardinality()/]"
                  xml:id="[aClass.name.concat('-').concat(anAttribute.name)/]">
        <xs:annotation>
    ['    <xs:documentation>' + anAttribute.u_commentBody().xsd_escapeXmlCharacters()/]</xs:documentation>
        </xs:annotation>
      </xs:element>
    [/for]
    [for (aProperty : uml::Property | aClass.u_associationEndProperties())]
      [let sourceClass : uml::Property = aProperty.association.ownedEnd->first()]
        [let targetClass : uml::Property = sourceClass.getOtherEnd()]
          <xs:element name="[aProperty.association.name/]"
                      minOccurs="[targetClass.lower/]"
                      maxOccurs="[targetClass.xsd_upperCardinality()/]"
                      xml:id="[aProperty.association.name/]">
            <xs:annotation>
    ['    <xs:documentation>' + aProperty.u_commentBody().xsd_escapeXmlCharacters()/]</xs:documentation>
            </xs:annotation>
            <xs:complexType>
              <xs:complexContent>
                <xs:restriction base="AssociationReferenceXsdType">
                  <xs:sequence>
                    <xs:element name="ddiReference"
                                type="InternationalRegistrationDataIdentifierXsdType"
                                minOccurs="0"
                                maxOccurs="1"
                                xml:id="[aProperty.association.name + '-ddiReference'/]"/>
                    <xs:element name="validType"
                                minOccurs="[sourceClass.lower/]"
                                maxOccurs="[sourceClass.xsd_upperCardinality()/]"
                                xml:id="[aProperty.association.name + '-validType'/]">
                      <xs:simpleType>
                        <xs:restriction base="xs:NMTOKEN">
                          [targetClass.type.xsd_referenceClass()/]
                        </xs:restriction>
                      </xs:simpleType>
                    </xs:element>
                  </xs:sequence>
                </xs:restriction>
              </xs:complexContent>
            </xs:complexType>
          </xs:element>
        [/let]
      [/let]
    [/for]
  </xs:sequence>
[/template]

[template public classProfileXsd(aClass : uml::Class, aProfilePackage : uml::Package)]
  [let subsetClasses : OrderedSet(uml::Class) =
      aProfilePackage.u_childDependencySupplierClasses()]
  [let superInSubset : uml::Class =
      aClass.u_allSuperClasses()->select(sc | subsetClasses->includes(sc))->first()]
    <xs:element name="[aClass.name/]"
                type="[aClass.name/]XsdType"
                xml:id="[aClass.name/]">
      <!-- based on the UML class [aClass.qualifiedName/] -->
      <xs:annotation>
  ['      <xs:documentation>' + aClass.u_commentBody().xsd_escapeXmlCharacters()/]</xs:documentation>
      </xs:annotation>
    </xs:element>
    <xs:complexType name="[aClass.name/]XsdType"
                    xml:id="[aClass.name/]XsdType">
      <xs:annotation>
  ['      <xs:documentation>' + aClass.u_commentBody().xsd_escapeXmlCharacters()/]</xs:documentation>
      </xs:annotation>
      [if (superInSubset <> null)]
        <xs:complexContent>
          <xs:extension base="[superInSubset.name/]XsdType">
            [if (aClass.u_attributes()->select(a | a.u_isAttributeInProfile(aClass, aProfilePackage))->notEmpty()
                or aClass.u_associationEndPropertiesInSubset(aProfilePackage)->notEmpty())]
              [aClass.propertyProfileXsd(aProfilePackage)/]
            [/if]
          </xs:extension>
        </xs:complexContent>
      [else]
        [if (aClass.u_attributes()->select(a | a.u_isAttributeInProfile(aClass, aProfilePackage))->notEmpty()
            or aClass.u_associationEndPropertiesInSubset(aProfilePackage)->notEmpty())]
          [aClass.propertyProfileXsd(aProfilePackage)/]
        [/if]
      [/if]
    </xs:complexType>
  [/let]
  [/let]
[/template]

[query public u_associationEndPropertiesInSubset(
  aClass : uml::Class,
  aProfilePackage : uml::Package) : Sequence(uml::Property) =
  let subsetClasses : OrderedSet(uml::Class) =
    aProfilePackage.u_childDependencySupplierClasses()
  in
  aClass.ownedAttribute
    ->select(p | p.association <> null
      and subsetClasses->includes(p.association.u_ownedEndClass())
      and subsetClasses->includes(p.association.u_otherEndClass()))
/]
