[template public xsd_referenceClass(aType : uml::Type) post(self.trim())]
  [let aClass : uml::Class = aType.oclAsType(uml::Class)]
    [let concretes : OrderedSet(uml::Class) = aClass.u_allConcreteSubClassesPlusSelf()]
      [if (concretes->notEmpty())]
        [for (aSubClass : uml::Class | concretes)]
          <xs:enumeration value="[aSubClass.name/]"/>
        [/for]
      [else]
        <xs:enumeration value="[aClass.name/]"/>
      [/if]
    [/let]
  [/let]
[/template]

[query public xsd_typeName(aType : uml::Type) : String =
  let n : String = aType.name in
  if n = 'Boolean' then
    'xs:boolean'
  else if n = 'Integer' then
    'xs:integer'
  else if n = 'String' then
    'xs:string'
  else if n = 'Real' then
    'xs:double'
  else if aType.package.name = 'XMLSchemaDataTypes' then
    aType.name.xsd_primitive()
  else
    n.concat('XsdType')
  endif
  endif
  endif
  endif
  endif
/]

[query public xsd_primitive(
  aTypeName : String) : String =
  if aTypeName = 'XsdAnyUri' then 'xs:anyURI'
  else if aTypeName = 'XsdDate' then 'xs:date'
  else if aTypeName = 'XsdLanguage' then 'xs:language'
  else 'WARNING: unresolved type'
  endif
  endif
  endif
/]

[query public xsd_escapeXmlCharacters(
  aString : String) : String =
  aString
  .replaceAll('&', '&amp;')
  .replaceAll('<', '&lt;')
  .replaceAll('>', '&gt;')
  .replaceAll('"', '&quot;')
  .replaceAll('\'', '&apos;')
