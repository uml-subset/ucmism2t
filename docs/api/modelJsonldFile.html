[template public generateJsonld(aModel : uml::Model)]
  [file (aModel.u_modelShortName() + '/encoding/json-ld/' + aModel.u_universePathSegment() + aModel.u_modelAcronym().toLower() + '.jsonld', overwrite, 'UTF-8')]
    [let classes : Sequence(uml::Class) =
            aModel.eAllContents(uml::Class)
            ->select(aClass | aClass.u_isInAllowedPackage()),
         datatypes : Sequence(uml::DataType) =
            aModel.eAllContents(uml::DataType)
            ->select(d | d.isComplex() and d.u_isInAllowedPackage() and d.isNotInMappedItems() and d.name <> 'PackageInformation')
      ]
      {
        "@context": {
          "[aModel.u_modelRdfPrefix()/]": "[aModel.u_modelRdfNamespace()/]",
[comment]         "['RDF_PREFIX'.getPropertyValue()/]": "['RDF_NAMESPACE'.getPropertyValue()/]",[/comment]
          "xsd": "http://www.w3.org/2001/XMLSchema#",
          "[aModel.name/]": "@included",
          [for (aClass : uml::Class | classes) separator(',\n    ')]
            "[aClass.name/]": {
              "@id": "[aClass.iri()/]",
              "@context": ['['/]
                [aClass.classJsonldSelfContained()/][aClass.classJsonldFile()/]
              ]
            }[/for]
          [if (datatypes->size()>0)],[/if]
          [for (aDataType : uml::DataType | datatypes) separator(',\n    ')]
            "[aDataType.name/]": {
              "@id": "[aDataType.iri()/]",
              "@context": ['['/]
                [aDataType.dataTypeJsonldSelfContained()/][aDataType.dataTypeJsonldFile()/]
              ]
            }[/for]
        },
        "generatedBy": "[aModel.generationMessageJsonld()/]",
        "basedOn": "based on the UML model [aModel.qualifiedName/]"
      }
    [/let]
  [/file]
[/template]

[template public profileGenerateJsonld(aProfilePackage : uml::Package)]
  [let aModel : uml::Model = aProfilePackage.u_model()]
    [let subsetClasses : OrderedSet(uml::Class) =
        aProfilePackage.u_childDependencySupplierClasses()]
    [let subsetDataTypes : OrderedSet(uml::DataType) =
        aProfilePackage.u_childDependencySupplierDataTypes()]
      [file (aModel.u_modelShortName() + '/encoding/json-ld/' + 'profiles_package'.getRequiredPropertyValue() + '/' + aProfilePackage.name + '/' + aModel.u_modelAcronym().toLower() + '.jsonld', overwrite, 'UTF-8')]
        {
          "@context": {
            "[aProfilePackage.u_modelRdfPrefix()/]": "[aProfilePackage.u_modelRdfNamespace()/]",
[comment]         "['RDF_PREFIX'.getPropertyValue()/]": "['RDF_NAMESPACE'.getPropertyValue()/]",[/comment]
            "xsd": "http://www.w3.org/2001/XMLSchema#",
            "[aModel.name/]": "@included",
            [for (aClass : uml::Class | subsetClasses->sortedBy(c | c.name)) separator(',
    ')]
              "[aClass.name/]": {
                "@id": "[aClass.iri()/]",
                "@context": ['['/]
                  [aClass.classJsonldSelfContainedProfile(aProfilePackage)/]
                ]
              }[/for]
            [if (subsetDataTypes->select(d | d.isComplex() and d.isNotInMappedItems())->size() > 0)],[/if]
            [for (aDataType : uml::DataType | subsetDataTypes->select(d | d.isComplex() and d.isNotInMappedItems() and d.name <> 'PackageInformation')->sortedBy(d | d.name)) separator(',
    ')]
              "[aDataType.name/]": {
                "@id": "[aDataType.iri()/]",
                "@context": ['['/]
                  [aDataType.dataTypeJsonldSelfContained()/]
                ]
              }[/for]
          },
          "generatedBy": "[aModel.generationMessageJsonld()/]",
          "basedOn": "based on UML model [aModel.qualifiedName/] profile [aProfilePackage.name/]"
        }
      [/file]
    [/let]
    [/let]
  [/let]
[/template]
