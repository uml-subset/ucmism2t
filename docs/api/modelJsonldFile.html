[template public generateJsonld(aModel : uml::Model)]
  [file (aModel.e_modelShortName() + '/encoding/json-ld/' + aModel.e_modelAcronym().toLower() + '.jsonld', overwrite, 'UTF-8')]
    {
      "@context": {
        "['RDF_PREFIX'.getPropertyValue()/]": "['RDF_NS'.getPropertyValue()/]",
     	"xsd": "http://www.w3.org/2001/XMLSchema#",
        "[aModel.name/]": "@included",
    [for (aClass : uml::Class | aModel.eAllContents(uml::Class)->select(aClass | aClass.u_isInAllowedPackage()))]
          "[aClass.name/]": {
            "@id": "[aClass.iri()/]",
            "@context": ['['/]
                [aClass.classJsonldSelfContained()/][aClass.classJsonldFile()/]
            ]
          },
    [/for]
    [for (aDataType : uml::DataType | aModel.eAllContents(uml::DataType)->select(aDataType | aDataType.isComplex() and aDataType.u_isInAllowedPackage() and aDataType.isNotInMappedItems())) separator(',\n')]
          "[aDataType.name/]": {
            "@id": "[aDataType.iri()/]",
            "@context": ['['/]
                [aDataType.dataTypeJsonldSelfContained()/][aDataType.dataTypeJsonldFile()/]
            ]
          }[/for]
    
      },
      "generatedBy": "[aModel.generationMessageJsonld()/]",
      "basedOn": "based on the UML model [aModel.qualifiedName/]"
    }
  [/file]
[/template]
