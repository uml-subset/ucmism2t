[template protected genAssociations(aClass : uml::Class, seen : OrderedSet(String)) post(self.trim())]
  [let associations : OrderedSet(uml::Association) = aClass.u_associations()->select(a | a.u_ownedEndClass() = aClass)]
    [for (anAssociation : uml::Association | associations->select(anAssociation | seen->excludes(anAssociation.name)))]
      [let sourceEnd : uml::Property = anAssociation.ownedEnd->first()]
        [let targetEnd : uml::Property = sourceEnd.getOtherEnd()]
          "[anAssociation.shortName()/]": {
            "description": "[anAssociation.name/]",
            "$comment": "cardinality [targetEnd.lowerBound()/]..[targetEnd.upperBound()/]",
          [if (targetEnd.upperBound() = 1)]
               "$ref": "#/defs/target-[anAssociation.u_otherEndClass().name/]"
          [else]
              "anyOf": ['['/]
                {
                  "$ref": "#/defs/target-[anAssociation.u_otherEndClass().name/]"
                },
                {
                  "type": "array",
                  "items": {
                    "$ref": "#/defs/target-[anAssociation.u_otherEndClass().name/]"
                  }
                }
              ]
          [/if]
          },
        [/let]
      [/let]
    [/for]
    [if (not aClass.superClass->isEmpty())]
      [aClass.superClass->asSequence()->collect(temp2 | temp2.genAssociations(seen + associations->asSequence()->collect(a | a.shortName())))/]
    [/if]
  [/let]
[/template]

[query public requiredClassAttributes(aClass : uml::Class) : String =
  let s : String = aClass.requiredAttributes()
  in
  if (aClass.superClass->isEmpty())
    s
  else
    s + aClass.superClass->first().requiredClassAttributes()
  endif

[query public requiredAssociations(aClass : uml::Class) : String =
  let s : String =
    aClass.u_associations()
      ->select(a1 | a1.u_ownedEndClass() = aClass)
      ->select(a2 | a2.u_otherEnd().lower > 0)
      ->collect(a3 | '"' + a3.shortName() + '", ')
      ->toString()
  in  
  if (aClass.superClass->isEmpty())
    s
  else
    s + aClass.superClass->first().requiredAssociations()
  endif
/]
