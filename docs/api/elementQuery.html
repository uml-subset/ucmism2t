[query public u_shortDocumentation(
  anElement : uml::Element) : String =
  let d : String = anElement.u_commentBody() in
  if d = '' then
    'WARNING: element has no documentation'
  else if d.size() > 100 then
    d.substring(1, 100) + '\n...'
  else
    d
  endif
  endif

[query public u_onlyDefinition(
  aNamedElement : uml::NamedElement) : String =
  let s : String =
    aNamedElement.u_commentBody()
      .replace('Explanatory notes[\\s\\S\\"=]+', '')
      .replace('Examples[\\s\\S\\"=]+', '')
      .replace('Definition[\\s\\S]+?[\\"=]+\\s+', '')
      .replaceAll('[\\s\\n\\r]+', ' ')
      .replaceAll('\\{', '<U+007B>')
      .replaceAll('\\}', '<U+007D>')
      .trim()
  in
  if s.size() > 300 then
    s.substring(1, 300).trim().concat(' ...')
  else
    s
  endif
/]

[query public u_isInAllowedPackage(
  aNamedElement : uml::NamedElement) : Boolean =
  aNamedElement <> null
  and aNamedElement.u_model()
    .u_mainPackage()
    .eAllContents(uml::NamedElement)
    ->includes(aNamedElement)

[query public u_isNotInIgnoredPackage(

[query public u_filePath(
  aNamedElement : uml::NamedElement,
  aTarget : String) : String =
  let root : String = aNamedElement.u_model().e_modelShortName()
    + '/' + aTarget + '/'
  in
  let path : String =
    aNamedElement.qualifiedName.replaceAll('::', '/')
  in
  if aTarget = 'doc' then
    root + path.replace(aNamedElement.u_model().name, '_rst')
  else
    root + path
  endif
/]
