[template public processClassifier(aClassifier : uml::Classifier)]
  [file (aClassifier.u_filePath('doc').concat('_xsd.txt'), overwrite, 'UTF-8')]
    [if (aClassifier.oclIsKindOf(uml::Class))]
      [aClassifier.oclAsType(uml::Class).classXsd()/]
    [elseif (aClassifier.oclIsKindOf(uml::Enumeration))]
      [aClassifier.oclAsType(uml::Enumeration).enumerationXsd()/]
    [elseif (aClassifier.oclIsKindOf(uml::DataType))]
      [aClassifier.oclAsType(uml::DataType).dataTypeXsd()/]
    [/if]
  [/file]
  [file (aClassifier.u_filePath('doc').concat('.schema.json'), overwrite, 'UTF-8')]
    [if (aClassifier.oclIsKindOf(uml::Class))]
      [aClassifier.oclAsType(uml::Class).classJsch()/]
    [elseif (aClassifier.oclIsKindOf(uml::Enumeration))]
      [aClassifier.oclAsType(uml::Enumeration).enumerationJsch()/]
    [elseif (aClassifier.oclIsKindOf(uml::DataType))]
      [aClassifier.oclAsType(uml::DataType).dataTypeJsch()/]
    [/if]
  [/file]
  [file (aClassifier.u_filePath('doc').concat('.onto.ttl'), overwrite, 'UTF-8')]
    [if (aClassifier.oclIsKindOf(uml::Class))]
      [aClassifier.oclAsType(uml::Class).classOnto()/]
    [elseif (aClassifier.oclIsKindOf(uml::Enumeration))]
      [aClassifier.oclAsType(uml::Enumeration).enumerationOnto()/]
    [elseif (aClassifier.oclIsKindOf(uml::DataType))]
      [aClassifier.oclAsType(uml::DataType).dataTypeOnto()/]
    [/if]
  [/file]
  [file (aClassifier.u_filePath('doc').concat('.jsonld'), overwrite, 'UTF-8')]
    [if (aClassifier.oclIsKindOf(uml::Class))]
      [aClassifier.oclAsType(uml::Class).classJsonldModular()/]
    [elseif (aClassifier.oclIsKindOf(uml::Enumeration))]
    [elseif (aClassifier.oclIsKindOf(uml::DataType))]
      [aClassifier.oclAsType(uml::DataType).dataTypeJsonldModular()/]
    [/if]
  [/file]
  [file (aClassifier.u_filePath('doc').concat('.shacl.ttl'), overwrite, 'UTF-8')]
    [if (aClassifier.oclIsKindOf(uml::Class))]
      [aClassifier.oclAsType(uml::Class).classShacl()/]
    [elseif (aClassifier.oclIsKindOf(uml::DataType))]
      [aClassifier.oclAsType(uml::DataType).dataTypeShacl()/]
    [/if]
  [/file]
  [file (aClassifier.u_filePath('doc').concat('.shexc'), overwrite, 'UTF-8')]
    [if (aClassifier.oclIsKindOf(uml::Class))]
      [aClassifier.oclAsType(uml::Class).classShex()/]
    [elseif (aClassifier.oclIsKindOf(uml::Enumeration))]
    [elseif (aClassifier.oclIsKindOf(uml::DataType))]
      [aClassifier.oclAsType(uml::DataType).dataTypeShex()/]
    [/if]
  [/file]
  
  [aClassifier.name.rst_header('=')/]
  
  [let shortDescription : String =
  	aClassifier.u_commentBody()
  	  .substring(aClassifier.u_commentBody().index('Description:') + 15)]
    .. dropdown:: [aClassifier.name/] - [shortDescription.substring(1, shortDescription.index('.'))/]
       :open:
  [/let]
  
     [aClassifier.u_commentBody()/]
  
     .. grid:: 2
        :gutter: 1
        :margin: 1 1 1 1
        :padding: 0 0 0 0
  
        .. grid-item-card:: Encodings Card 1
  
           [aClassifier.encodingTabSet('1', ':selected:', '')/]
  
        .. grid-item-card:: Encodings Card 2
  
           [aClassifier.encodingTabSet('2', '', ':selected:')/]
  
[/template]
