[template public packageOntoFile(aPackage : uml::Package)]
  [let isModelRoot : Boolean = aPackage.eContainer(uml::Package) = null]
  [file (aPackage.u_model().u_modelShortName() + '/encoding/ontology/' + (if isModelRoot then '' else aPackage.u_universePathSegment() endif) + aPackage.name + '.onto.ttl', overwrite, 'UTF-8')]
    # PLEASE DO NOT EDIT THIS CODE!
    # This code was generated by the Eclipse Acceleo module UCMIS M2T.
    # Target language is 'Ontology Turtle' ('onto.ttl') generated on the basis of the model '[aPackage.u_model().name/]'.
    
    # Ontology for [aPackage.name/]
    # based on the UML package [aPackage.qualifiedName/]
    [aPackage.u_model().ttl_prefixes()/]
    
    <[aPackage.name/].onto.ttl> a owl:Ontology;
      rdfs:label "[aPackage.name/]";
      [aPackage.rdfs_comment()/]
    [for (subPackage : uml::Package | aPackage.u_childPackages()
      ->select(p1 | p1.u_isAllowedPackage())
      ->select(p2 | 'IGNORE_PACKAGE'.getPropertyValue().substituteAll(' ', '').tokenize(',')->excludes(s | s=aPackage.name))
      ->select(p3 | not ('profiles'.isPropertyTrue() and (p3.name = 'profiles_package'.getRequiredPropertyValue() or p3.eContainer(uml::Package).name = 'profiles_package'.getRequiredPropertyValue())))
      )]
        owl:imports <[if (isModelRoot)][subPackage.u_universePathSegment()/][/if][subPackage.name/].onto.ttl>;[subPackage.packageOntoFile()/]
    [/for]
    .
    
    [for (aClass : uml::Class | aPackage.u_childClasses())]
      [aClass.classOnto()/]
    [/for]
    
    [for (aDataType : uml::DataType | aPackage.u_childrenDataTypes()
      ->select(aDataType | aDataType.isComplex() and aDataType.isNotInMappedItems() and aDataType.name <> 'PackageInformation'))]
      [aDataType.dataTypeOnto()/]
    [/for]
    
    [for (anEnumeration : uml::Enumeration | aPackage.u_childrenEnumerations())]
      [anEnumeration.enumerationOnto()/]
    [/for]
  [/file]
  [/let]
[/template]

[template public profileOntoFile(aProfilePackage : uml::Package)]
  [let aModel : uml::Model = aProfilePackage.u_model()]
    [let subsetClasses : OrderedSet(uml::Class) =
        aProfilePackage.u_childDependencySupplierClasses()]
    [let subsetDataTypes : OrderedSet(uml::DataType) =
        aProfilePackage.u_childDependencySupplierDataTypes()]
      [file (aModel.u_modelShortName() + '/encoding/ontology/' + 'profiles_package'.getRequiredPropertyValue() + '/' + aProfilePackage.name + '/' + aProfilePackage.name + '.onto.ttl', overwrite, 'UTF-8')]
        # PLEASE DO NOT EDIT THIS CODE!
        # This code was generated by the Eclipse Acceleo module UCMIS M2T.
        # Target language is 'Ontology Turtle' ('onto.ttl').
        
        # Ontology for profile [aProfilePackage.name/]
        [aModel.ttl_prefixes()/]
        
        <[aProfilePackage.name/].onto.ttl> a owl:Ontology;
          rdfs:label "[aProfilePackage.name/]";
        .
        
        [for (aClass : uml::Class | subsetClasses->sortedBy(c | c.name))]
          [aClass.classProfileOnto(aProfilePackage)/]
        [/for]
        
        [for (aDataType : uml::DataType | subsetDataTypes->sortedBy(d | d.name)
            ->select(d | d.isComplex() and d.isNotInMappedItems() and d.name <> 'PackageInformation'))]
          [aDataType.dataTypeOnto()/]
        [/for]
      [/file]
    [/let]
    [/let]
  [/let]
[/template]
