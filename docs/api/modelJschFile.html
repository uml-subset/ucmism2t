[template public profileGenerateJsonSchema(aProfilePackage : uml::Package)]
  [let aModel : uml::Model = aProfilePackage.u_model()]
    [let subsetClasses : OrderedSet(uml::Class) =
        aProfilePackage.u_childDependencySupplierClasses()]
    [let subsetDataTypes : OrderedSet(uml::DataType) =
        aProfilePackage.u_childDependencySupplierDataTypes()]
      [file (aModel.u_modelShortName() + '/encoding/json-schema/' + 'profiles_package'.getRequiredPropertyValue() + '/' + aProfilePackage.name + '/' + aModel.u_modelAcronym().toLower() + '.schema.json', overwrite, 'UTF-8')]
        {
          "$schema": "https://json-schema.org/draft/2020-12/schema",
          "title": "[aModel.name/] - [aProfilePackage.name/]",
          "description": "This code was generated by the Eclipse Acceleo module UCMIS M2T.",
          "version": "[aModel.jschVersion()/]",
        
          "if": { "type": "object" },
          "then": { "anyOf": ['['/]
            { "$ref": "#/defs/root-object" },
            { "$ref": "#/defs/root-graph" }
          ]},
          "else": { "$ref": "#/defs/root-array" },
        
          "defs": {
            "root-object": {
              "type": "object",
              "required": ['['/]"@type"],
              "anyOf": ['['/]
                [aProfilePackage.arrayClassSchemasProfile()/]
                false
              ]
            },
            "root-array": {
              "type": "array",
              "items": { "$ref": "#/defs/root-object" }
            },
            "root-graph": {
              "type": "object",
              "required": ['['/]"[aModel.name/]"],
              "properties": {
                "[aModel.name/]": { "$ref": "#/defs/root-array" }
              }
            },
        [for (aClass : uml::Class | subsetClasses->sortedBy(c | c.name))]
              [aClass.classProfileJsch(aProfilePackage)/]
        [/for]
        [for (anEnumeration : uml::Enumeration | subsetDataTypes
            ->select(d | d.oclIsTypeOf(uml::Enumeration))
            ->collect(d | d.oclAsType(uml::Enumeration))
            ->sortedBy(e | e.name))]
              [anEnumeration.enumerationJsch()/]
        [/for]
        [for (aDataType : uml::DataType | subsetDataTypes
            ->select(d | d.isComplex() and d.isNotInMappedItems()
              and d.name <> 'PackageInformation')
            ->sortedBy(d | d.name))]
              [aDataType.dataTypeJsch()/]
        [/for]
            "xsd:string": { "type": "string" },
            "xsd:boolean": { "type": "boolean" },
            "xsd:integer": { "type": "integer" },
            "xsd:double": { "type": "number" },
            "xsd:nonNegativeInteger": { "type": "integer", "minumum": 0 },
            "xsd:anyURI": { "type": "string", "format": "uri-reference" },
            "xsd:date": { "type": "string", "format": "date" },

                { "type": "object" }
              ]
            },

              "anyOf": ['['/]
                { "type": "string", "format": "iri-reference" },
                {
                  "type": "array",
                  "items": {
                    "anyOf": ['['/]
                      { "type": "string", "format": "iri-reference" },
                      { "type": "object" }
                    ]
                  }
                }
              ]
            }
          }
        }
      [/file]
    [/let]
