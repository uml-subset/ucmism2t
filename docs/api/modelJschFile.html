[template public generateJsonSchema(aModel : uml::Model)]
  [file (aModel.e_modelShortName() + '/encoding/json-schema/' + aModel.e_modelAcronym().toLower() + '.schema.json', overwrite, 'UTF-8')]
    {
      "$schema": "https://json-schema.org/draft/2020-12/schema",
      "title": "[aModel.name/]",
      "description": "This code was generated by the Eclipse Acceleo module UCMIS M2T.",
      "version": "[aModel.jschVersion()/]",
    
      "if": { "type": "object" },
      "then": { "anyOf": ['['/]
        { "$ref": "#/defs/root-object" },
        { "$ref": "#/defs/root-graph" }
      ]},
      "else": { "$ref": "#/defs/root-array" },
    
      "defs": {
        "root-object": {
          "type": "object",
          "required": ['['/]"@type"],
          "anyOf": [aModel.arrayClassSchemas()/]
        },
        "root-array": {
          "type": "array",
          "items": { "$ref": "#/defs/root-object" }
        },
        "root-graph": {
          "type": "object",
          "required": ['['/]"[aModel.name/]"],
          "properties": {
            "[aModel.name/]": { "$ref": "#/defs/root-array" }
          }
        },
    [for (aClass : uml::Class | aModel.u_mainPackage().eAllContents(uml::Class)->sortedBy(temp1 | temp1.name))]
          [aClass.classJsch()/]
    [/for]
    [for (anEnumeration : uml::Enumeration | aModel.u_mainPackage().eAllContents(uml::Enumeration)->sortedBy(temp2 | temp2.name))]
          [anEnumeration.enumerationJsch()/]
    [/for]
    [for (aDataType : uml::DataType | aModel.eAllContents(uml::DataType)->select(aDataType | aDataType.isComplex() and aDataType.u_isInAllowedPackage() and aDataType.isNotInMappedItems()))]
          [aDataType.dataTypeJsch()/]
    [/for]
